# Warning and sanitizer flags apply to xombe's own targets only, so a project that adds xombe with
# add_subdirectory() keeps its own flags.
function(xombe_target_options target)
  if(CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
    # no a*b+c -> FMA contraction (clang's default), so output is bit-identical with and without FMA hardware
    target_compile_options(${target} PRIVATE -Wall -Wextra -ffp-contract=off)
    if(XOMBE_WERROR)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  elseif(MSVC)
    target_compile_options(${target} PRIVATE /W4)
    if(XOMBE_WERROR)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  endif()
  if(XOMBE_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer
                                             -fno-sanitize-recover=all)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()

function(xombe_library_options target)
  target_include_directories(${target}
    PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
           $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
           $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE src)
  # MSVC's C runtime has the math functions, and there is no libm
  if(NOT MSVC)
    target_link_libraries(${target} PUBLIC m)
  endif()
  xombe_target_options(${target})
endfunction()

function(xombe_executable name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} xombe)
  xombe_target_options(${name})
endfunction()

if(XOMBE_INSTALL)
  install(TARGETS xombe EXPORT xombeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
          LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES include/xombe.h ${CMAKE_CURRENT_BINARY_DIR}/include/xombe_config.h
          DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(EXPORT xombeTargets NAMESPACE xombe:: FILE xombeConfig.cmake
          DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xombe)
  include(CMakePackageConfigHelpers)
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/xombeConfigVersion.cmake
                                   COMPATIBILITY SameMinorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/xombeConfigVersion.cmake
          DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xombe)
  # xombe.pc finds the prefix from its own directory, so it holds under cmake --install --prefix. an absolute
  # CMAKE_INSTALL_LIBDIR or INCLUDEDIR stays absolute
  if(IS_ABSOLUTE ${CMAKE_INSTALL_LIBDIR})
    set(XOMBE_PC_PREFIX ${CMAKE_INSTALL_PREFIX})
  else()
    file(RELATIVE_PATH XOMBE_PC_PREFIX ${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig ${CMAKE_INSTALL_PREFIX})
    string(REGEX REPLACE "/$" "" XOMBE_PC_PREFIX "\${pcfiledir}/${XOMBE_PC_PREFIX}")
  endif()
  foreach(d LIBDIR INCLUDEDIR)
    if(IS_ABSOLUTE ${CMAKE_INSTALL_${d}})
      set(XOMBE_PC_${d} ${CMAKE_INSTALL_${d}})
    else()
      set(XOMBE_PC_${d} "\${prefix}/${CMAKE_INSTALL_${d}}")
    endif()
  endforeach()
  configure_file(xombe.pc.in ${CMAKE_CURRENT_BINARY_DIR}/xombe.pc @ONLY)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/xombe.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
  if(XOMBE_BUILD_TOOLS)
    install(TARGETS xombe-dec RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    if(XOMBE_ENCODER)
      install(TARGETS xombe-enc RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
    if(XOMBE_DSTAR)
      install(TARGETS xombe-xc RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
  endif()
endif()
